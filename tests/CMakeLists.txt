cmake_minimum_required(VERSION 3.16)
project(wifi_manager_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(ENABLE_SANITIZERS "Address and undefined sanitizers" OFF)
option(ENABLE_COVERAGE "Coverage instrumentation" OFF)
if(ENABLE_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
if(ENABLE_COVERAGE)
 add_compile_options(--coverage -O0 -g)
 add_link_options(--coverage)
 if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  # Adapter tests update shared-function counters from multiple threads.
  # Non-atomic gcov updates can produce impossible negative hit counts.
  add_compile_options(-fprofile-update=atomic)
 endif()
endif()
enable_testing()
add_library(check_c_header OBJECT test_header.c)
target_include_directories(check_c_header PRIVATE c_api ../include)
target_compile_options(check_c_header PRIVATE -Wall -Wextra -Werror)
add_executable(test_policy test_policy.cpp ../src/policy.cpp)
target_include_directories(test_policy PRIVATE ../src)
target_compile_options(test_policy PRIVATE -Wall -Wextra -Werror)
add_test(NAME policy COMMAND test_policy)
find_package(Threads REQUIRED)
option(ENABLE_TSAN "Thread sanitizer" OFF)
if(ENABLE_TSAN)
 if(ENABLE_SANITIZERS)
  message(FATAL_ERROR "TSAN cannot be combined with ASAN")
 endif()
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
endif()
add_executable(test_adapter test_adapter.cpp fakes/fake.cpp ../src/wifi_manager.cpp ../src/policy.cpp)
target_include_directories(test_adapter PRIVATE fakes ../include ../src)
target_link_libraries(test_adapter PRIVATE Threads::Threads)
target_compile_options(test_adapter PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME adapter COMMAND test_adapter)
add_executable(test_adapter_coarse test_adapter.cpp fakes/fake.cpp ../src/wifi_manager.cpp ../src/policy.cpp)
target_include_directories(test_adapter_coarse PRIVATE fakes ../include ../src)
target_compile_definitions(test_adapter_coarse PRIVATE FAKE_TICK_PERIOD_MS=10)
target_link_libraries(test_adapter_coarse PRIVATE Threads::Threads)
target_compile_options(test_adapter_coarse PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME adapter_100hz COMMAND test_adapter_coarse)
